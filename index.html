            </div>

    <script>

        const inputField = document.getElementById('word-input');

        const searchBtn = document.getElementById('search-btn');

        const resultContainer = document.getElementById('result');


        // Trigger search when clicking the button

        searchBtn.addEventListener('click', fetchDictionaryData);


        // Trigger search when pressing 'Enter' on the keyboard

        inputField.addEventListener('keypress', function(e) {

            if (e.key === 'Enter') {

                fetchDictionaryData();

            }

        });


        async function fetchDictionaryData() {

            const word = inputField.value.trim();

            

            // Don't search if the input is empty

            if (!word) return;


            // Show loading state

            resultContainer.innerHTML = `<p class="loading">Searching for "${word}"...</p>`;


            try {

                // Fetch data from the Free Dictionary API

                const response = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${word}`);

                

                if (!response.ok) {

                    throw new Error("Word not found");

                }


                const data = await response.json();

                displayResult(data[0]);


            } catch (error) {

                // Handle errors (like word not existing)

                resultContainer.innerHTML = `

                    <div class="error">

                        <p>Oops! We couldn't find the word "${word}".</p>

                        <p style="font-size: 0.9rem; margin-top: 5px;">Please check the spelling and try again.</p>

                    </div>

                `;

            }

        }


        function displayResult(data) {

            // Extract the word and phonetic spelling

            const word = data.word;

            const phonetic = data.phonetic || (data.phonetics.find(p => p.text) ? data.phonetics.find(p => p.text).text : "");


            let html = `

                <div class="word-header">

                    <h2 class="word-title">${word}</h2>

                </div>

                ${phonetic ? `<p class="phonetic">${phonetic}</p>` : ''}

            `;


            // Loop through the different parts of speech (noun, verb, etc.)

            data.meanings.forEach(meaning => {

                html += `

                    <div class="meaning-group">

                        <p class="part-of-speech">${meaning.partOfSpeech}</p>

                        <ul class="definitions">

                `;


                // Limit to the first 3 definitions per part of speech to keep it clean

                meaning.definitions.slice(0, 3).forEach(def => {

                    html += `<li>${def.definition}</li>`;

                });


                html += `

                        </ul>

                    </div>

                `;

            });


            // Update the webpage with the generated HTML

            resultContainer.innerHTML = html;

        }

    </script>
